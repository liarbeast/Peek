{# A field's element. A field dragged to a place gets its box as inline style: position, size, front-to-back order, alignment and, for text, its size. #} {% macro field(e) -%} {%- set V = {"top": "flex-start", "middle": "center", "bottom": "flex-end"} -%} {%- if e.value -%} {%- set b = e.box -%} {%- set classes = "field field-" ~ e.key ~ " type-" ~ e.type ~ (" placed" if b else "") -%} {%- if b -%} {%- set align = b.align or "center" -%} {%- set valign = b.valign or "middle" -%} {%- set style = "left:%dpx;top:%dpx;width:%dpx;height:%dpx" % (b.x, b.y, b.w, b.h) -%} {%- if b.z is number %}{% set style = style ~ ";z-index:%d" % b.z %}{% endif -%} {%- if e.type == "image" -%} {%- set style = style ~ ";object-position:%s %s" % (align, {"top": "top", "middle": "center", "bottom": "bottom"}[valign]) -%} {%- else -%} {%- set style = style ~ ";text-align:%s;justify-content:%s" % (align, V[valign]) -%} {%- if b.size %}{% set style = style ~ ";font-size:%dpx" % b.size %}{% endif -%} {%- endif -%} {%- endif -%} {%- if e.type == "image" -%} {%- elif b -%}
{{ e.value }}
{%- else -%}
{{ e.value }}
{%- endif -%} {%- endif -%} {%- endmacro %}
