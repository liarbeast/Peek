{# A field's element. A field dragged to a place gets its box as inline style: position, front-to-back order and alignment; an image also gets its size, while text gets its width and font size and is as tall as its lines. #} {% macro field(e) -%} {%- if e.value -%} {%- set b = e.box -%} {%- set classes = "field field-" ~ e.key ~ " type-" ~ e.type ~ (" placed" if b else "") -%} {%- if b -%} {%- set align = b.align or "center" -%} {%- set style = "left:%dpx;top:%dpx;width:%dpx" % (b.x, b.y, b.w) -%} {%- if e.type == "image" -%} {%- set valign = {"top": "top", "middle": "center", "bottom": "bottom"}[b.valign or "middle"] -%} {%- set style = style ~ ";height:%dpx;object-position:%s %s" % (b.h, align, valign) -%} {%- else -%} {%- set style = style ~ ";text-align:%s" % align -%} {%- if b.size %}{% set style = style ~ ";font-size:%dpx" % b.size %}{% endif -%} {%- endif -%} {%- if b.z is number %}{% set style = style ~ ";z-index:%d" % b.z %}{% endif -%} {%- endif -%} {%- set link = ' data-below="%s" data-gap="%d"' % (b.below | e, b.gap or 0) if b and b.below else "" -%} {%- if e.type == "image" -%} {%- else -%}
{{ e.value }}
{%- endif -%} {%- endif -%} {%- endmacro %}
