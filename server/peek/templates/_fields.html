{# Draws one field's value with its classes: field field- type-. A field dragged to a place gets that box, outside the normal flow. #} {% macro field(e) -%} {%- if e.value -%} {%- set classes = "field field-" ~ e.key ~ " type-" ~ e.type ~ (" placed" if e.box else "") -%} {%- set style = "left:%dpx;top:%dpx;width:%dpx;height:%dpx" % (e.box.x, e.box.y, e.box.w, e.box.h) if e.box else "" -%} {%- if e.type == "image" -%} {%- else -%}
{{ e.value }}
{%- endif -%} {%- endif -%} {%- endmacro %}
